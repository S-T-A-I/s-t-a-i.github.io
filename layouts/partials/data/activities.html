{{- /* Fetches social activities and invited talks at BUILD time from the internal tool, where the lab records them. See data/members.html. Returns a slice; empty slice on any failure. */ -}} {{- $out := slice -}} {{- $url := .Site.Params.activities_api_url -}} {{- if $url -}} {{- $t := try (resources.GetRemote $url) -}} {{- if $t.Err -}} {{- warnf "STAI: activities API unreachable (%s): %s" $url $t.Err -}} {{- else -}} {{- with $t.Value -}} {{- $j := . | transform.Unmarshal -}} {{- with $j.activities }}{{ $out = . }}{{ end -}} {{- end -}} {{- end -}} {{- end -}} {{- return $out -}}
