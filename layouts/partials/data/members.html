{{- /* Fetches the members list at BUILD time so every name is in the HTML source (search engines, Scholar, link previews) and the page still renders if the assets API is down at request time. Returns a slice; empty slice on any failure. */ -}} {{- $out := slice -}} {{- $url := .Site.Params.members_api_url -}} {{- if $url -}} {{- $t := try (resources.GetRemote $url) -}} {{- if $t.Err -}} {{- warnf "STAI: members API unreachable (%s): %s" $url $t.Err -}} {{- else -}} {{- with $t.Value -}} {{- $j := . | transform.Unmarshal -}} {{- with $j.members }}{{ $out = . }}{{ end -}} {{- end -}} {{- end -}} {{- end -}} {{- return $out -}}
